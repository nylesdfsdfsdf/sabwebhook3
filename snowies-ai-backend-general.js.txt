import express from "express";
import "dotenv/config";

const app = express();
app.use(express.json({ limit: "64kb" }));

const OPENAI_API_KEY = process.env.OPENAI_API_KEY;

if (!OPENAI_API_KEY) {
  console.error("Missing OPENAI_API_KEY environment variable.");
  process.exit(1);
}

const SYSTEM_PROMPT = `
You are Snowies AI Helper, a helpful general-purpose AI assistant built into the Snowies Admin Panel.

You can answer general questions across a wide range of topics, including programming, Roblox development, writing, school subjects, technology, explanations, brainstorming, and everyday questions.

When Snowies Admin Panel context is supplied for the current request, use that context to answer panel-related questions accurately.

Do not unnecessarily mention Snowies, roles, whitelists, teleporting, Join Sounds, or the admin panel when the user's question is unrelated to the panel.

Treat panel context as factual context supplied by the Roblox server. Never claim you performed an admin action unless the system actually confirms that action happened.

Be clear, helpful, and easy to understand.
`;

function cleanHistory(value) {
  if (!Array.isArray(value)) return [];

  return value.slice(-10).flatMap((item) => {
    if (!item || typeof item !== "object") return [];

    const role = item.role === "assistant" ? "assistant" : "user";

    const text =
      typeof item.text === "string"
        ? item.text.trim().slice(0, 1200)
        : "";

    return text ? [{ role, text }] : [];
  });
}

function questionLooksPanelRelated(question, panelContext) {
  const q = question.toLowerCase();

  const panelTerms = [
    "snowies",
    "admin panel",
    "admin",
    "owner",
    "co-owner",
    "coowner",
    "whitelist",
    "whitelisted",
    "access",
    "join sound",
    "join sounds",
    "teleport",
    "protected",
    "protection",
    "role",
    "roles",
    "theme",
    "themes",
    "owner settings",
    "how long left",
    "permanent access",
  ];

  if (panelTerms.some((term) => q.includes(term))) {
    return true;
  }

  const players = Array.isArray(panelContext?.currentPlayers)
    ? panelContext.currentPlayers
    : [];

  return players.some((player) => {
    const username =
      typeof player?.username === "string"
        ? player.username.toLowerCase()
        : "";

    const displayName =
      typeof player?.displayName === "string"
        ? player.displayName.toLowerCase()
        : "";

    return (
      (username && q.includes(`@${username}`)) ||
      (username && q.includes(username)) ||
      (displayName &&
        displayName.length >= 3 &&
        q.includes(displayName))
    );
  });
}

function panelContextText(panelContext, settings) {
  if (!panelContext || typeof panelContext !== "object") {
    return "";
  }

  const requester = panelContext.requester ?? {};

  const players = Array.isArray(panelContext.currentPlayers)
    ? panelContext.currentPlayers
    : [];

  const lines = [
    "SNOWIES ADMIN PANEL CONTEXT:",
    `Requester: @${requester.username ?? "unknown"}`,
    `Requester role: ${requester.role ?? "unknown"}`,
    `Requester access remaining: ${requester.access ?? "unknown"}`,
    `Join Sounds target: ${
      panelContext.joinSoundTarget
        ? `@${panelContext.joinSoundTarget}`
        : "none"
    }`,
  ];

  if (settings && typeof settings === "object") {
    if (
      typeof settings.theme === "string" &&
      settings.theme
    ) {
      lines.push(`Current UI theme: ${settings.theme}`);
    }

    if (typeof settings.uiScale === "number") {
      lines.push(`Current UI scale: ${settings.uiScale}`);
    }
  }

  if (players.length) {
    lines.push("Current server players:");

    for (const player of players.slice(0, 30)) {
      lines.push(
        `- @${player.username ?? "unknown"} | ` +
          `display=${player.displayName ?? "unknown"} | ` +
          `role=${player.role ?? "none"} | ` +
          `access=${player.access ?? "No Access"} | ` +
          `protected=${player.protected ? "yes" : "no"} | ` +
          `permanent=${player.permanent ? "yes" : "no"}`
      );
    }
  }

  return lines.join("\n");
}

function buildInput({
  question,
  history,
  mode,
  panelContext,
  settings,
}) {
  const input = [];

  for (const item of history) {
    input.push({
      role: item.role,
      content: item.text,
    });
  }

  const usePanelContext = questionLooksPanelRelated(
    question,
    panelContext
  );

  let currentMessage = question;

  if (usePanelContext) {
    const context = panelContextText(
      panelContext,
      settings
    );

    if (context) {
      currentMessage =
        `${context}\n\n` +
        `USER QUESTION:\n${question}`;
    }
  }

  if (mode === "Short") {
    currentMessage +=
      "\n\nResponse preference: keep the answer short and concise.";
  } else if (mode === "Detailed") {
    currentMessage +=
      "\n\nResponse preference: give a detailed but focused explanation.";
  } else {
    currentMessage +=
      "\n\nResponse preference: give a normal, useful level of detail.";
  }

  input.push({
    role: "user",
    content: currentMessage,
  });

  return input;
}

app.get("/", (_req, res) => {
  res.json({
    ok: true,
    service: "Snowies AI Helper",
  });
});

app.post("/ask", async (req, res) => {
  try {
    const question =
      typeof req.body?.question === "string"
        ? req.body.question.trim().slice(0, 1000)
        : "";

    if (!question) {
      return res.status(400).json({
        success: false,
        message: "Question is required.",
      });
    }

    const mode =
      req.body?.mode === "Short" ||
      req.body?.mode === "Detailed"
        ? req.body.mode
        : "Normal";

    const history = cleanHistory(req.body?.history);

    const panelContext =
      req.body?.panelContext &&
      typeof req.body.panelContext === "object"
        ? req.body.panelContext
        : null;

    const settings =
      req.body?.settings &&
      typeof req.body.settings === "object"
        ? req.body.settings
        : {};

    const input = buildInput({
      question,
      history,
      mode,
      panelContext,
      settings,
    });

    const response = await fetch(
      "https://api.openai.com/v1/responses",
      {
        method: "POST",

        headers: {
          "Content-Type": "application/json",
          Authorization: `Bearer ${OPENAI_API_KEY}`,
        },

        body: JSON.stringify({
          model: "gpt-5.6-luna",
          instructions: SYSTEM_PROMPT,
          input,
          max_output_tokens: 700,
        }),
      }
    );

    const data = await response.json();

    if (!response.ok) {
      console.error("OpenAI error:", data);

      const apiMessage =
        typeof data?.error?.message === "string"
          ? data.error.message
          : "OpenAI request failed.";

      return res.status(502).json({
        success: false,
        message: apiMessage,
      });
    }

    let answer = "";

    for (const item of data.output ?? []) {
      for (const content of item.content ?? []) {
        if (
          content.type === "output_text" &&
          typeof content.text === "string"
        ) {
          answer += content.text;
        }
      }
    }

    answer = answer.trim();

    return res.json({
      success: true,
      answer: answer || "No response.",
    });
  } catch (error) {
    console.error("Backend error:", error);

    return res.status(500).json({
      success: false,
      message: "Server error.",
    });
  }
});

const port = Number(process.env.PORT || 3000);

app.listen(port, () => {
  console.log(
    `Snowies AI Helper general backend running on port ${port}`
  );
});
